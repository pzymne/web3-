pragma solidity ^0.8.0;

contract BinarySearch {
    /**
     * @dev 在升序数组中执行二分查找
     * @param arr 有序数组（必须按升序排列）
     * @param target 要查找的目标值
     * @return 目标值在数组中的索引，如果未找到则返回 type(uint256).max
     */
    function binarySearch(uint256[] memory arr, uint256 target) 
        public 
        pure 
        returns (uint256) 
    {
        // 定义搜索区间的左边界（包含）
        uint256 left = 0;
        
        // 定义搜索区间的右边界（不包含）
        uint256 right = arr.length;
        
        // 当搜索区间不为空时继续查找
        // left < right 表示区间至少包含一个元素
        while (left < right) {
            // 计算中间位置，使用防溢出公式
            // 标准公式 (left + right) / 2 可能溢出
            // 这里使用 left + (right - left) / 2 避免溢出
            uint256 mid = left + (right - left) / 2;
            
            // 获取中间位置的元素值
            uint256 midValue = arr[mid];
            
            // 如果中间值等于目标值，找到目标，返回索引
            if (midValue == target) {
                return mid;
            }
            
            // 如果中间值小于目标值，说明目标在右半部分
            // 更新左边界为 mid + 1（因为 mid 已经检查过）
            else if (midValue < target) {
                left = mid + 1;
            }
            
            // 如果中间值大于目标值，说明目标在左半部分
            // 更新右边界为 mid（因为 mid 已经检查过）
            else {
                right = mid;
            }
        }
        
        // 如果循环结束仍未找到目标值，返回一个特殊值表示未找到
        // type(uint256).max 是 uint256 的最大值，通常不会作为有效索引
        return type(uint256).max;
    }
}
